import { createElement as h, useState, version } from 'react';
import { createRoot } from 'react-dom/client';
import { flushSync, version as domVersion } from 'react-dom';
import { RadioGroup } from '@base-ui/react/radio-group';
import { Radio } from '@base-ui/react/radio';
import type { Ref } from 'react';
export function mountProbe(host: HTMLElement, initial: Ref<HTMLInputElement>) {
  let replace: (next: Ref<HTMLInputElement>) => void = () => {};
  function Fixture() {
    const [inputRef, setInputRef] = useState(() => initial);
    replace = next => setInputRef(() => next);
    return h(RadioGroup, {defaultValue: 'a',inputRef},h(Radio.Root, {value: 'a',...{'data-testid': 'a'}}),h(Radio.Root,{value: 'b',...{'data-testid':'b'}}));
  }
  const root = createRoot(host);
  flushSync(() => root.render(h(Fixture)));
  return {version,domVersion,replace: (next: Ref<HTMLInputElement>) => flushSync(() => replace(next)),unmount:()=>flushSync(()=>root.unmount())};
}
