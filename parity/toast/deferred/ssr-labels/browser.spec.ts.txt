function readLabelRelationships(doc: Document) {
  return [...doc.querySelectorAll('[data-label-case]')].map(root => ({
    mode: root.getAttribute('data-label-case'),
    labelled: root.getAttribute('aria-labelledby'), described: root.getAttribute('aria-describedby'),
    titles: [...root.querySelectorAll('h2')].map(title => title.id), description: root.querySelector('p')?.id ?? null,
  }));
}
function expectLabelRelationships(records: ReturnType<typeof readLabelRelationships>) {
  expect(records).toHaveLength(8);
  const ids = records.flatMap(record => [...record.titles, ...(record.description ? [record.description] : [])]);
  expect(new Set(ids).size).toBe(ids.length);
  for (const record of records) {
    if (record.mode?.endsWith('empty')) {
      expect(record.labelled).toBeNull(); expect(record.described).toBeNull(); expect(record.titles).toEqual([]);
    } else {
      expect(record.labelled).toBe(record.titles.at(-1)); expect(record.labelled).toBeTruthy();
      expect(record.described).toBe(record.description); expect(record.described).toBeTruthy();
    }
  }
}
test('supplement: SSR label relationships name native Roots with JavaScript disabled', async ({ browser }) => {
  const context = await browser.newContext({ javaScriptEnabled: false });
  try {
    const page = await context.newPage(); await page.goto('http://127.0.0.1:5173/toast?case=ssr-labels');
    await expect(page.locator('main')).toHaveAttribute('data-hydrated', 'false');
    expectLabelRelationships(await page.evaluate(readLabelRelationships));
    await expect(page.getByRole('dialog', { name: 'Title', exact: true })).toHaveCount(4);
    await expect(page.getByRole('dialog', { name: 'Second title', exact: true })).toHaveCount(2);
  } finally { await context.close(); }
});
test('supplement: SSR generated custom and duplicate labels survive hydration and conditional remounts', async ({ page }) => {
  const errors: string[] = []; page.on('pageerror', error => errors.push(error.message));
  const response = await page.goto('/toast?case=ssr-labels');
  const html = await response!.text();
  const server = await page.evaluate(html => {
    const doc = new DOMParser().parseFromString(html, 'text/html');
    return [...doc.querySelectorAll('[data-label-case]')].map(root => ({
      mode: root.getAttribute('data-label-case'),
      labelled: root.getAttribute('aria-labelledby'), described: root.getAttribute('aria-describedby'),
      titles: [...root.querySelectorAll('h2')].map(title => title.id), description: root.querySelector('p')?.id ?? null,
    }));
  }, html);
  expectLabelRelationships(server);
  await expect(page.locator('main')).toHaveAttribute('data-hydrated', 'true');
  expect(await page.evaluate(readLabelRelationships)).toEqual(server);
  await click(page, 'hide labels');
  for (const root of await page.locator('[data-label-case]').all()) {
    await expect(root).not.toHaveAttribute('aria-labelledby'); await expect(root).not.toHaveAttribute('aria-describedby');
  }
  await click(page, 'show labels'); expectLabelRelationships(await page.evaluate(readLabelRelationships));
  expect(errors).toEqual([]);
});
