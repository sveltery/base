<script lang="ts">
  // Svelte binding settlement publishes descendant label IDs before final SSR output.
  import { setContext, type Snippet } from 'svelte';
  import Element from '../dialog/Element.svelte';
  import { ROOT, type ToastRootContext } from './root-context.js';
  let { controller, idPrefix, titleId = $bindable(), descriptionId = $bindable(), internal,
    props, state, children, ref = $bindable(null), attach }: {
    controller: Pick<ToastRootContext, 'toast' | 'expanded' | 'visibleIndex' | 'recalculateHeight'>;
    idPrefix: string; titleId?: string; descriptionId?: string;
    internal: Record<string, unknown>; props: Record<string, unknown>; state: Record<string, unknown>;
    children?: Snippet; ref?: HTMLElement | null; attach: (node: HTMLElement) => () => void;
  } = $props();
  const counts = { title: 0, description: 0 };
  let titleRegistration: object | undefined;
  let descriptionRegistration: object | undefined;
  const context: ToastRootContext = {
    get toast() { return controller.toast; },
    get expanded() { return controller.expanded; },
    get visibleIndex() { return controller.visibleIndex; },
    get titleId() { return titleId; },
    get descriptionId() { return descriptionId; },
    allocateLabelId(part) { return `${idPrefix}-${part}-${counts[part]++}`; },
    setTitleId(id) {
      const registration = {}; titleRegistration = registration; titleId = id;
      return () => { if (titleRegistration === registration) titleId = undefined; };
    },
    setDescriptionId(id) {
      const registration = {}; descriptionRegistration = registration; descriptionId = id;
      return () => { if (descriptionRegistration === registration) descriptionId = undefined; };
    },
    recalculateHeight(flush) { controller.recalculateHeight(flush); },
  };
  setContext(ROOT, context);
</script>
<Element {internal} {props} {state} {children} {attach} bind:ref />
