<script lang="ts">
  import { Toast } from '@sveltery/base';
  let showLabels = $state(true);
</script>
<button onclick={() => showLabels = false}>hide labels</button>
<button onclick={() => showLabels = true}>show labels</button>
{#each [0, 1] as provider (provider)}
  <Toast.Provider><Toast.Viewport>
    {#each ['generated', 'custom', 'duplicate', 'empty'] as mode (mode)}
      <Toast.Root toast={{ id: `${provider}-${mode}`, title: mode === 'empty' ? null : 'Title', description: mode === 'empty' ? false : 'Description' }} swipeDirection={[]} data-label-case={`${provider}-${mode}`}>
        {#if showLabels}
          <Toast.Title id={mode === 'custom' ? `title-${provider}` : undefined} />
          {#if mode === 'duplicate'}<Toast.Title>Second title</Toast.Title>{/if}
          <Toast.Description id={mode === 'custom' ? `description-${provider}` : undefined} />
        {/if}
      </Toast.Root>
    {/each}
  </Toast.Viewport></Toast.Provider>
{/each}
